# Grouped by the pipeline phase that first needs them (see the project
# brief's task list). Versions here are minimums, not exact pins.
# requirements.lock.txt is the exact, reproducible resolution of this
# file (all platforms, Python >= 3.11) -- install from it for a
# byte-for-byte environment, or from this file to get the latest
# compatible versions. Regenerate it after editing this file with:
#   uv pip compile requirements.txt --universal --python-version 3.11 -o requirements.lock.txt
#
# Only the Phase 0 group is installed as part of scaffolding; the rest are
# listed now so the full shape of the environment is visible up front, but
# are meant to be installed when you actually start that phase (several of
# them -- spacy's model download, sentence-transformers' first model pull,
# shap's native build -- are slow and better done deliberately, not as a
# side effect of `pip install -r requirements.txt`).

# --- Phase 0: scaffolding ---
# langgraph is listed here since the state schema (state.py) is a Phase 0
# artifact, but it wasn't actually installed/needed until Phase 7 -- every
# agent from Phase 1-6 is a standalone function operating on a plain dict,
# not a wired-up graph node, until pipeline.py connects them.
langgraph>=0.2
python-dotenv>=1.0
pydantic>=2.7

# --- Phase 2: Document Parser Agent ---
# pdfplumber, not Marker/Docling as originally pitched -- both of those are
# ML-based parsers for messy scanned documents and pull in torch as a
# transitive dependency. This MVP's PDFs are synthetic and born-digital
# (see data/synthetic_patients/), so a non-ML text/table extractor is a
# better fit; see the module docstring in agents/document_parser.py.
pdfplumber>=0.11
pydicom>=2.4
numpy>=1.26  # pydicom needs it for .pixel_array

# --- Phase 3: Privacy Protection Agent ---
presidio-analyzer>=2.2
presidio-anonymizer>=2.2
spacy>=3.7
# after installing spacy: python -m spacy download en_core_web_lg
# (~587MB.) This is the model Presidio's default AnalyzerEngine() actually
# loads -- an earlier version of this comment said en_core_web_sm, but the
# code never configured that; without _lg installed, Presidio silently
# downloads it at first use (found while setting up CI). Every test and
# live run to date used _lg, so it stays; switching to the ~15MB _sm would
# lower NER recall on names and needs measuring first.

# --- Phase 4: Medical Knowledge RAG Agent ---
chromadb>=0.5
# No sentence-transformers, no biopython -- see agents/medical_knowledge_rag.py's
# module docstring. chromadb's bundled DefaultEmbeddingFunction already uses
# an ONNX MiniLM model via onnxruntime (which chromadb installs on its own),
# so a second ML runtime (torch) isn't needed for "local embeddings, no
# per-call API cost." PubMed fetching uses the E-utilities REST API directly
# via the standard library -- biopython's Entrez module is a thin wrapper
# over the same two HTTP endpoints.

# --- Phase 5: Diagnostic Prediction Agent ---
# Routed through OpenRouter's free router (openrouter/free), reached with
# the `openai` package pointed at OpenRouter's base URL -- OpenRouter is
# OpenAI-API-compatible, so no separate SDK is needed. Three providers
# were tried in order before landing here: Gemini (google-genai) hit a
# persistent, unresolvable API_KEY_SERVICE_BLOCKED account-permission
# error; OpenAI called directly has no meaningful free tier (billing
# required even for trial credit); a local Ollama model was ruled out
# because a deployed app can't reach a model running on someone's laptop.
# See the module docstring in agents/diagnostic_prediction.py for the
# full account.
openai>=1.50
tenacity>=8.2  # retry/backoff around the OpenRouter call specifically (Phase 7)
pillow>=10.0  # diagnostic_prediction.py converts DICOM pixel data to PNG for the multimodal prompt; was only ever present as a transitive dependency of other packages

# --- Phase 6: Explainability Agent ---
shap>=0.45
scikit-learn>=1.4
pandas>=2.2

# --- Phase 8: UI ---
chainlit>=1.1

# --- Security floors for transitive dependencies ---

cryptography>=50.0
